--@name QPix Formats
--@author StyledStrike
--@client

local formats = {}

-- Grayscale, each shade is stored in 1 byte as a grayscale value (0-255)
formats["GRAYSCALE"] = {
    readFullFrame = function( reader, framePixelCount, changed, outR, outG, outB )
        local ReadUInt8 = reader.ReadUInt8
        local color = 0

        for i = 1, framePixelCount do
            color = ReadUInt8()

            changed[i] = i - 1
            outR[i] = color * 255
            outG[i] = color * 255
            outB[i] = color * 255
        end
    end,

    readDeltaFrame = function( reader, changedPixelCount, changed, outR, outG, outB )
        local ReadUInt8 = reader.ReadUInt8
        local ReadUInt16 = reader.ReadUInt16
        local pixelI, color = 0, 0

        for i = 1, changedPixelCount do
            pixelI = ReadUInt16()
            color = ReadUInt8()

            changed[i] = pixelI
            outR[i] = color * 255
            outG[i] = color * 255
            outB[i] = color * 255
        end
    end
}

-- RGB color, stored in three bytes as `RRRRRRRR`, `GGGGGGGG` and `BBBBBBBB`
formats["RGB888"] = {
    readFullFrame = function( reader, framePixelCount, changed, outR, outG, outB )
        local ReadUInt8_8_8 = reader.ReadUInt8_8_8

        for i = 1, framePixelCount do
            changed[i] = i - 1
            outR[i], outG[i], outB[i] = ReadUInt8_8_8()
        end
    end,

    readDeltaFrame = function( reader, changedPixelCount, changed, outR, outG, outB )
        local ReadUInt8_8_8 = reader.ReadUInt8_8_8
        local ReadUInt16 = reader.ReadUInt16

        for i = 1, changedPixelCount do
            changed[i] = ReadUInt16()
            outR[i], outG[i], outB[i] = ReadUInt8_8_8()
        end
    end
}

local Band = bit.band
local Brshift = bit.rshift

-- RGB color, stored in a byte as `RRRGGGBB` (3 bits for red, 3 bits for green, 2 bits for blue)
formats["PACKED_RGB332"] = {
    readFullFrame = function( reader, framePixelCount, changed, outR, outG, outB )
        local ReadUInt8 = reader.ReadUInt8
        local packed = 0

        for i = 1, framePixelCount do
            packed = ReadUInt8()

            changed[i] = i - 1
            outR[i] = ( Band( Brshift( packed, 5 ), 0x07 ) / 7 ) * 255 -- (packed & 0b11100000) >> 5
            outG[i] = ( Band( Brshift( packed, 2 ), 0x07 ) / 7 ) * 255 -- (packed & 0b00011100) >> 2
            outB[i] = ( Band( packed, 0x03 ) / 3 ) * 255               -- (packed & 0b00000011)
        end
    end,

    readDeltaFrame = function( reader, changedPixelCount, changed, outR, outG, outB )
        local ReadUInt8 = reader.ReadUInt8
        local ReadUInt16 = reader.ReadUInt16
        local pixelI, packed = 0, 0

        for i = 1, changedPixelCount do
            pixelI = ReadUInt16()
            packed = ReadUInt8()

            changed[i] = pixelI
            outR[i] = ( Band( Brshift( packed, 5 ), 0x07 ) / 7 ) * 255 -- (packed & 0b11100000) >> 5
            outG[i] = ( Band( Brshift( packed, 2 ), 0x07 ) / 7 ) * 255 -- (packed & 0b00011100) >> 2
            outB[i] = ( Band( packed, 0x03 ) / 3 ) * 255               -- (packed & 0b00000011)
        end
    end
}

-- RGB color, stored in two bytes as `RRRRGGGG` and `0000BBBB` (4 bits for red, 4 bits for green, 4 bits for blue)
formats["PACKED_RG44_B4"] = {
    readFullFrame = function( reader, framePixelCount, changed, outR, outG, outB )
        local ReadUInt8_8 = reader.ReadUInt8_8
        local color, b = 0, 0

        for i = 1, framePixelCount do
            color, b = ReadUInt8_8()

            changed[i] = i - 1
            outR[i] = ( Brshift( color, 4 ) / 15 ) * 255 -- color >> 4
            outG[i] = ( Band( color, 15 ) / 15 ) * 255   -- color & 0b00001111
            outB[i] = ( b / 15 ) * 255
        end
    end,

    readDeltaFrame = function( reader, changedPixelCount, changed, outR, outG, outB )
        local ReadUInt8_8 = reader.ReadUInt8_8
        local ReadUInt16 = reader.ReadUInt16
        local pixelI, color, b = 0, 0, 0

        for i = 1, changedPixelCount do
            pixelI = ReadUInt16()
            color, b = ReadUInt8_8()

            changed[i] = pixelI
            outR[i] = ( Brshift( color, 4 ) / 15 ) * 255 -- color >> 4
            outG[i] = ( Band( color, 15 ) / 15 ) * 255   -- color & 0b00001111
            outB[i] = ( b / 15 ) * 255
        end
    end
}

return formats
